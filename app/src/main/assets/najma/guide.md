# مرحباً بالعالم
> أول سكربت لك: ماذا تكتب وأين، وكيف تشغّله.

نجمة لغة برمجة عربية لصنع ألعابك في Star2D. تكتب الأوامر بالعربية، والمحرك يحوّلها إلى Java قبل تشغيل اللعبة، فتعمل بنفس سرعة Java تماماً.

## أين أكتب الكود؟

- سكربت العنصر: اختر عنصراً في المشهد، افتح الأحداث، ثم Body Script، واختر «نجمة (عربي)».
- سكربت المشهد: من الأحداث افتح Scene Script واختر «نجمة (عربي)».

## أول برنامج

هذا سكربت مشهد يعرض رسالة عند بدء اللعبة:

```مشهد
عند البدء {
    اكتب "مرحباً بالعالم!"
}
```

احفظ ثم شغّل اللعبة، وستظهر الرسالة أسفل الشاشة.

## قواعد الكتابة

- كل أمر في سطر لوحده.
- الأوامر التي تحتوي أوامر أخرى (مثل الأحداث والشروط) تضع ما بداخلها بين قوسين { }.
- السطر الذي يبدأ بـ # ملاحظة لك، يتجاهلها المحرك.
- الأرقام العربية ١٢٣ والإنجليزية 123 كلاهما مقبول.
- التشكيل والهمزات و«ال» التعريف لا تهم: اللاعب و لاعب نفس الاسم.

```مشهد
# هذه ملاحظة ولن تُنفذ
عند البدء {
    اكتب "السطر الأول"
    اكتب "السطر الثاني"
}
```

! اكتب أول حرفين من أي أمر وستظهر لك فقاعة الاقتراحات. اضغط على الاقتراح ليُكتب كاملاً.

# المتغيرات والأنواع
> صناديق تحفظ فيها الأرقام والنصوص وتغيّرها أثناء اللعب.

المتغير صندوق له اسم وبداخله قيمة. تعرّفه بكلمة «متغير»، والمحرك يعرف نوعه من القيمة.

```جزء
متغير النقاط = ٠
متغير السرعة = ٢٫٥
متغير الاسم = "سارة"
متغير حي = صح
اكتب "اللاعب {الاسم} عنده {النقاط} نقطة"
```

## تغيير القيمة

```جزء
متغير النقاط = ٠
اجعل النقاط = ١٠
زد النقاط بمقدار ٥
انقص النقاط بمقدار ٣
زد النقاط
انقص النقاط
اكتب "النقاط الآن {النقاط}"
```

«زد» و«انقص» بدون «بمقدار» تغيّر القيمة بواحد.

## الأنواع

إذا أردت تحديد النوع بنفسك فاكتبه بدل كلمة «متغير»:

```جزء
رقم_صحيح العمر = ١٢
رقم_عشري الطول = ١٫٤٥
نص المدينة = "الخرطوم"
منطقي فاز = خطأ
حرف الدرجة = 'أ'
رقم_طويل رقم_كبير = ٩٠٠٠٠٠٠٠٠٠
عشري_دقيق باي_دقيق = ٣٫١٤١٥٩٢٦٥٣٥
```

- رقم_صحيح: رقم بلا كسور مثل ٥ و -٣.
- رقم_عشري: رقم بكسور مثل ٢٫٥.
- نص: كلام بين علامتي تنصيص.
- منطقي: صح أو خطأ.

## أين أعرّف المتغير؟

المتغير المعرّف خارج الأحداث يعيش طوال اللعبة وتراه كل الأحداث. المعرّف داخل حدث يختفي عند نهاية ذلك الحدث.

```عنصر
متغير عدد_النقرات = ٠

عند النقر {
    زد عدد_النقرات
    اكتب "نقرت {عدد_النقرات} مرة"
}
```

## الخاصية والثابت

«خاصية» متغير يظهر في لوحة Properties لتغيّر قيمته بدون فتح الكود. «ثابت» قيمة لا تتغير أبداً.

```عنصر
خاصية السرعة = ٢٠٠
ثابت أقصى_حياة = ٣

عند البدء {
    اكتب "السرعة {السرعة} والحياة {أقصى_حياة}"
}
```

# الحساب والنصوص
> الجمع والطرح والقسمة، ودمج النصوص والتحويل بينها وبين الأرقام.

## العمليات

```جزء
متغير أ = ١٠
متغير ب = ٣
اكتب "الجمع {أ + ب}"
اكتب "الطرح {أ - ب}"
اكتب "الضرب {أ × ب}"
اكتب "القسمة {أ ÷ ب}"
اكتب "باقي القسمة {أ باقي ب}"
```

- × أو * للضرب.
- ÷ قسمة حقيقية: ١٠ ÷ ٤ = ٢٫٥
- / قسمة مثل Java: ١٠ / ٤ = ٢ إذا كان الرقمان صحيحين.
- ٣٠٪ تعني ٠٫٣

## دوال الحساب

```جزء
اكتب "تقريب {قرب(٣٫٧)}"
اكتب "لأعلى {قرب_لاعلى(٣٫٢)} ولأسفل {قرب_لاسفل(٣٫٩)}"
اكتب "مطلق {مطلق(-٥)}"
اكتب "جذر {جذر(١٦)} وأس {اس(٢، ٣)}"
اكتب "الأكبر {الأكبر(٤، ٩)} والأصغر {الأصغر(٤، ٩)}"
اكتب "تحديد {حدد(١٥٠، ٠، ١٠٠)}"
```

«حدد» تبقي الرقم بين حدين، ومفيدة لمنع اللاعب من الخروج من الشاشة.

## النصوص

ضع اسم المتغير بين { } داخل النص ليظهر مكانه. ويمكنك دمج النصوص بعلامة +.

```جزء
متغير الاسم = "علي"
متغير تحية = "أهلاً " + الاسم
اكتب "{تحية}، طول اسمك {طول(الاسم)} حروف"
اكتب استبدل("قطة صغيرة"، "قطة"، "نمر")
اكتب اقطع("مرحباً بالعالم"، ٠، ٦)
إذا يبدا_ب(الاسم، "ع") {
    اكتب "اسمك يبدأ بحرف العين"
}
إذا تحية تحتوي "علي" {
    اكتب "التحية فيها علي"
}
```

## التحويل بين الأنواع

```جزء
متغير مكتوب = "٤٢"
متغير رقمي = رقم_صحيح(مكتوب) + ٨
متغير كسر = رقم_عشري("٢٫٥")
متغير كنص = نص(رقمي) + " نقطة"
متغير بدون_كسور = رقم_صحيح(٧٫٩)
اكتب "{رقمي} {كسر} {كنص} {بدون_كسور}"
```

# الشروط
> اجعل اللعبة تقرر: إذا حدث كذا افعل كذا، وإلا افعل شيئاً آخر.

```جزء
متغير النقاط = ٧٥
إذا النقاط أكبر من ٥٠ {
    اكتب "ممتاز!"
} وإلا {
    اكتب "حاول مرة أخرى"
}
```

## المقارنات

- أكبر من ، أصغر من ، أقل من
- أكبر من أو يساوي ، أصغر من أو يساوي
- يساوي ، لا يساوي
- بين رقمين: بين ١ و ١٠
- ويمكن كتابتها بالرموز: > < >= <= == !=

```جزء
متغير العمر = ١٢
إذا العمر بين ٧ و ١٢ {
    اكتب "في المرحلة الابتدائية"
}
إذا العمر لا يساوي ١٠ {
    اكتب "ليس عشر سنوات"
}
```

## أكثر من شرط

«و» تعني أن الشرطين صحيحان معاً، «أو» تعني أن أحدهما يكفي، و«ليس» تعكس الشرط.

```جزء
متغير النقاط = ٨٠
متغير الحياة = ٢
إذا النقاط أكبر من ٥٠ و الحياة أكبر من ٠ {
    اكتب "ما زلت في اللعبة"
}
إذا النقاط يساوي ١٠٠ أو الحياة يساوي ٣ {
    اكتب "مثالي"
}
إذا ليس (الحياة يساوي ٠) {
    اكتب "لم تخسر بعد"
}
```

## عدة حالات

```جزء
متغير الدرجة = ٨٥
إذا الدرجة أكبر من أو يساوي ٩٠ {
    اكتب "امتياز"
} وإلا إذا الدرجة أكبر من أو يساوي ٧٥ {
    اكتب "جيد جداً"
} وإلا إذا الدرجة أكبر من أو يساوي ٥٠ {
    اكتب "ناجح"
} وإلا {
    اكتب "راسب"
}
```

## حسب

عندما تقارن قيمة واحدة بعدة احتمالات، «حسب» أوضح من إذا المتكررة:

```جزء
متغير اليوم = ٣
حسب اليوم {
    حالة ١: اكتب "السبت"
    حالة ٢: اكتب "الأحد"
    حالة ٣، ٤: اكتب "وسط الأسبوع"
    الباقي: اكتب "يوم آخر"
}
```

## قيمة حسب شرط

```جزء
متغير النقاط = ٣٠
متغير النتيجة = النقاط أكبر من ٥٠ ? "فوز" : "خسارة"
اكتب النتيجة
```

# الحلقات
> كرّر الأوامر بدل كتابتها مرات كثيرة.

## كرر

```جزء
كرر ٣ مرات {
    اكتب "مرحباً"
}
كرر ٥ مرات باستخدام ع {
    اكتب "الدورة رقم {ع}"
}
```

العدّاد «ع» يبدأ من صفر: ٠ ثم ١ ثم ٢...

## لكل ... من ... إلى

```جزء
لكل ع من ١ إلى ٥ {
    اكتب "الرقم {ع}"
}
لكل ع من ١٠ إلى ٠ بخطوة -٢ {
    اكتب "عدّ تنازلي {ع}"
}
```

## لكل ... في

تمر على عناصر قائمة واحداً واحداً:

```جزء
متغير الفواكه = ["تفاح"، "موز"، "برتقال"]
لكل فاكهة في الفواكه {
    اكتب "أحب {فاكهة}"
}
```

## طالما

تكرر ما دام الشرط صحيحاً. تأكد أن الشرط سيصبح خطأ يوماً ما، وإلا توقفت اللعبة!

```جزء
متغير العد = ٥
طالما العد أكبر من ٠ {
    اكتب "باقي {العد}"
    انقص العد
}
```

## اخرج وتخطى

«اخرج» تنهي الحلقة فوراً، و«تخطى» تقفز إلى الدورة التالية.

```جزء
لكل ع من ١ إلى ١٠ {
    إذا ع يساوي ٣ { تخطى }
    إذا ع أكبر من ٦ { اخرج }
    اكتب "{ع}"
}
```

# القوائم والقواميس
> احفظ مجموعة قيم في متغير واحد: قائمة أسماء، قائمة أعداء، أسعار.

## القائمة

```جزء
متغير الأسماء = ["علي"، "سارة"، "محمد"]
أضف "ليلى" إلى الأسماء
احذف "علي" من الأسماء
اكتب "الأول {الأسماء[٠]} والعدد {طول(الأسماء)}"
اجعل الأسماء[٠] = "نور"
إذا الأسماء تحتوي "محمد" {
    اكتب "محمد موجود"
}
```

الترقيم يبدأ من صفر: أول عنصر رقمه ٠.

## قائمة بنوع محدد وقائمة فارغة

```جزء
قائمة من رقم_صحيح الدرجات = []
أضف ٩٠ إلى الدرجات
أضف ٧٥ إلى الدرجات
أضف ٦٠ إلى الدرجات
احذف العنصر رقم ٠ من الدرجات
اكتب "المجموع {مجموع(الدرجات)} والأكبر {اكبر_قيمة(الدرجات)}"
```

## ترتيب وتغيير القائمة

```جزء
متغير الأرقام = [٥، ٢، ٩، ١]
رتب الأرقام
اكتب "تصاعدي {الأرقام}"
رتب الأرقام تنازلياً
اكتب "تنازلي {الأرقام}"
اعكس الأرقام
اخلط الأرقام
اكتب "أول {اول(الأرقام)} وآخر {اخر(الأرقام)}"
اكتب "عشوائي {اختر_عشوائيا(الأرقام)}"
أفرغ الأرقام
```

## القاموس

القاموس يربط مفتاحاً بقيمة، مثل اسم السلعة وسعرها:

```جزء
قاموس من نص إلى رقم_صحيح الأسعار = جديد قاموس من نص إلى رقم_صحيح()
اجعل الأسعار["قهوة"] = ٥٠٠
اجعل الأسعار["عصير"] = ٧٠٠
زد الأسعار["قهوة"] بمقدار ١٠٠
اكتب "سعر القهوة {الأسعار["قهوة"]}"
إذا الأسعار تحتوي "شاي" {
    اكتب "عندنا شاي"
}
لكل سلعة في الأسعار {
    اكتب "{سلعة}: {الأسعار[سلعة]}"
}
```

## أنواع أخرى

```جزء
مصفوفة من رقم_صحيح الخانات = [٠، ٠، ٠]
اجعل الخانات[١] = ٧
مجموعة من نص الزوار = جديد مجموعة من نص()
الزوار.أضف("علي")
الزوار.أضف("علي")
اكتب "عدد الزوار بدون تكرار {طول(الزوار)}"
قائمة_مرتبطة من نص الطابور = جديد قائمة_مرتبطة من نص()
الطابور.أضف("الأول")
```

- مصفوفة: حجمها ثابت وهي الأسرع.
- مجموعة: لا تقبل التكرار.
- قائمة_مرتبطة ، طابور ، مكدس ، قاموس_مرتب: للمحترفين، وهي نفسها LinkedList و ArrayDeque و TreeMap في Java.

# المهام
> اجمع عدة أوامر تحت اسم واحد واستعملها متى شئت.

المهمة مجموعة أوامر لها اسم. تعرّفها مرة واحدة خارج الأحداث، ثم تناديها باسمها وبعده قوسان.

```عنصر
مهمة رحب() {
    اكتب "أهلاً بك في اللعبة"
}

عند البدء {
    رحب()
}
```

## مهمة بمدخلات

```عنصر
مهمة حيّ(نص الاسم، رقم_صحيح العمر) {
    اكتب "أهلاً {الاسم}، عمرك {العمر}"
}

عند البدء {
    حيّ("سارة"، ١٠)
    حيّ("علي"، ١٢)
}
```

المدخل المكتوب بلا نوع يُعتبر رقماً عشرياً.

## مهمة ترجع قيمة

```عنصر
مهمة مساحة(العرض، الارتفاع) {
    ارجع العرض × الارتفاع
}

مهمة أكبر_من_عشرة(رقم_صحيح ر): منطقي {
    ارجع ر أكبر من ١٠
}

عند البدء {
    متغير م = مساحة(٤، ٥)
    اكتب "المساحة {م}"
    إذا أكبر_من_عشرة(١٥) { اكتب "نعم" }
}
```

نوع القيمة الراجعة يُعرف تلقائياً، أو تكتبه بعد نقطتين كما في المهمة الثانية.

# الأحداث
> متى يعمل الكود: عند البدء، كل إطار، عند النقر، عند الاصطدام.

الحدث هو اللحظة التي يعمل فيها الكود. تكتب اسم الحدث ثم ما يجب فعله بين { }.

## أحداث سكربت العنصر

```عنصر
عند البدء {
    اكتب "العنصر ظهر في اللعبة"
}

كل إطار {
    # يعمل حوالي ٦٠ مرة في الثانية: مناسب للحركة
}

عند النقر {
    اكتب "نقرت عليّ"
}

عند بداية اللمس {
    اجعل حجم نفسي = ٠٫٩
}

عند نهاية اللمس {
    اجعل حجم نفسي = ١
}

عند الاصطدام {
    اكتب "اصطدمت بـ {اسم الآخر}"
}

عند نهاية الاصطدام {
    اكتب "ابتعدت عن {اسم الآخر}"
}
```

- «نفسي» هو العنصر صاحب السكربت.
- «الآخر» في أحداث الاصطدام هو العنصر الذي اصطدمت به.

## أحداث سكربت المشهد

```مشهد
عند البدء {
    اكتب "بدأ المشهد"
}

كل إطار {
}

عند الإيقاف {
    # خرج اللاعب من اللعبة مؤقتاً
}

عند الاستئناف {
    # عاد اللاعب
}
```

# عناصر المشهد
> اقرأ وغيّر موضع العناصر وحجمها ولونها ونصها.

كل عنصر في المشهد تكتبه باسمه مباشرة كما سمّيته في المحرر، مثل player و coin.

## الخصائص

تكتب اسم الخاصية ثم اسم العنصر: «س player» تعني موضع اللاعب الأفقي.

```جزء
اجعل س player = ١٠٠
اجعل ص player = ٣٠٠
زد س player بمقدار ٥٠
متغير عرضه = عرض player
اجعل دوران coin = ٤٥
اجعل شفافية enemy = ٠٫٥
اجعل حجم coin = ١٫٥
اجعل مرئي enemy = خطأ
اجعل لون player = أحمر
اجعل نص scoreText = "النقاط: ١٠"
اجعل تقدم bar = ٧٥
```

- س ، ص: الموضع بالبكسل (ص تزيد للأعلى).
- عرض ، ارتفاع: الحجم (للقراءة فقط).
- دوران: الزاوية بالدرجات.
- شفافية: من ٠ (مخفي) إلى ١ (ظاهر).
- حجم: ١ هو الحجم الطبيعي.
- مرئي: صح أو خطأ.
- سرعة_س ، سرعة_ص: السرعة بالبكسل في الثانية.
- اسم ، نص ، لون ، صورة ، تقدم ، حي.

يمكنك أيضاً كتابتها بالنقطة: player.س

## الألوان الجاهزة

أحمر ، أخضر ، أزرق ، أصفر ، أبيض ، أسود ، برتقالي ، بنفسجي ، وردي ، رمادي ، سماوي ، بني ، شفاف. أو لون خاص بنسب الأحمر والأخضر والأزرق والشفافية:

```جزء
اجعل لون player = أزرق
اجعل لون coin = لون(١، ٠٫٥، ٠، ١)
اجعل لون enemy = "#FF8800"
```

```جزء
أخفِ enemy
أظهر enemy
انقل coin إلى ٢٠٠ و ٤٠٠
دوّر coin بزاوية ١٠
```

# الدوال القصيرة والقوائم المتقدمة
> فلترة وبحث وعدّ داخل القوائم بسطر واحد.

الدالة القصيرة تُكتب هكذا: (المدخل) ← النتيجة

```جزء

متغير كم_صغير = عد(الأرقام، (ر) ← ر أصغر من ١٠)

لكل_عنصر(الأرقام، (ر) ← اطبع(ر))
```

```عنصر
            اكتب اقطع(الجملة، ٠، ظاهر) في نفسي
        }
    }
}
```

## ٢٦. رسم خطوط بـ libGDX

```مشهد
راسم_الأشكال الراسم
متغير النقاط_س = [١٠٠٫٠، ٣٠٠٫٠، ٥٠٠٫٠]
متغير النقاط_ص = [٢٠٠٫٠، ٦٠٠٫٠، ٢٠٠٫٠]

عند البدء {
    اجعل الراسم = جديد راسم_الأشكال()
}

    الراسم.ابدأ(خطي)
    اجعل اللون الراسم = أصفر
    لكل ع من ٠ إلى طول(النقاط_س) - ٢ {
        الراسم.خط(النقاط_س[ع]، النقاط_ص[ع]، النقاط_س[ع + ١]، النقاط_ص[ع + ١])
    }
    الراسم.أنهِ()
}
```
